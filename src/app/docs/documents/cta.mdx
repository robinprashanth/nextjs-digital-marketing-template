## Component Types

1. **CTACard** - Simple card-style CTA
2. **CTASection** - Full-width section CTA
3. **GradientCTA** - Gradient background CTA
4. **SupportCTA** - Support-focused CTA

## Basic Usage

```tsx
import { CTACard, CTASection, GradientCTA, SupportCTA } from '@/components/cta';

// Simple Card CTA
<CTACard
  title="Join Our Team"
  description="We're looking for talented individuals"
  buttonText="View Positions"
  path="/careers"
/>

// Section CTA
<CTASection
  title="Ready to Transform?"
  description="Let's discuss your goals"
  background="gradient"
  buttons={[
    {
      href: "/contact",
      text: "Get Started",
      variant: "white-action"
    }
  ]}
/>

// Gradient CTA
<GradientCTA
  title="Start Your Project"
  description="Transform your business"
  buttons={[
    {
      text: "Get Started",
      href: "/contact",
      variant: "primary"
    }
  ]}
  gradient="purple"
/>
```

## Component Props

### CTACard Props

```typescript
interface CTACardProps {
  title: string;
  description: string;
  buttonText: string;
  path?: string;
  variant?: "primary" | "custom";
  showArrow?: boolean;
  className?: string;
}
```

### CTASection Props

```typescript
interface CTASectionProps {
  title: string | ReactNode;
  description: string;
  background?: "primary" | "navy" | "gradient";
  className?: string;
  viewport?: boolean;
  buttons: CTAButtonProps[];
}

interface CTAButtonProps {
  href: string;
  text: string;
  variant?: ButtonVariant;
  size?: "xl" | "fluid";
  rounded?: "full";
  showArrow?: boolean;
}
```

### GradientCTA Props

```typescript
interface GradientCTAProps {
  title: string;
  description: string;
  buttons: CTAButton[];
  className?: string;
  gradient?: "purple" | "ocean" | "rose" | "sunset";
  useInView?: boolean;
}

interface CTAButton {
  text: string;
  href: string;
  variant: "primary" | "secondary";
}
```

## Styling Options

### Background Variants

```tsx
// Solid Background
<CTASection background="primary" />
<CTASection background="navy" />

// Gradient Background
<CTASection background="gradient" />
<GradientCTA gradient="purple" />
<GradientCTA gradient="ocean" />
<GradientCTA gradient="rose" />
<GradientCTA gradient="sunset" />
```

### Button Variants

```tsx
const buttons = [
  {
    variant: "white-action",
    size: "xl",
    rounded: "full",
    showArrow: true,
  },
  {
    variant: "glass",
    size: "fluid",
  },
  {
    variant: "cta2",
    rounded: "full",
  },
];
```
